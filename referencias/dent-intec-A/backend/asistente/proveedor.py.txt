import httpx
from django.conf import settings

# Sin connect timeout, httpx puede quedarse en IPv6 muerta (p. ej. VPS con AAAA pero sin salida v6).
def _openai_http_timeout() -> httpx.Timeout:
    lectura = float(settings.ASISTENTE_LLM_TIMEOUT_S)
    return httpx.Timeout(connect=10.0, read=lectura, write=60.0, pool=10.0)


def crear_modelo_chat(*, modelo=None):
    """Devuelve un chat model de LangChain (Ollama u OpenAI según LLM_SERVER)."""
    if modelo is not None:
        return modelo

    if settings.LLM_SERVER == 'openai':
        from langchain_openai import ChatOpenAI

        return ChatOpenAI(
            api_key=settings.OPENAI_API_KEY,
            model=settings.OPENAI_MODEL,
            reasoning_effort=settings.OPENAI_REASONING_EFFORT or None,
            max_tokens=settings.OPENAI_MAX_TOKENS,
            timeout=_openai_http_timeout(),
        )

    from langchain_ollama import ChatOllama

    return ChatOllama(
        base_url=settings.OLLAMA_BASE_URL,
        model=settings.OLLAMA_MODEL,
        timeout=float(settings.ASISTENTE_LLM_TIMEOUT_S),
        keep_alive=settings.OLLAMA_KEEP_ALIVE,
        num_ctx=settings.OLLAMA_NUM_CTX,
    )
