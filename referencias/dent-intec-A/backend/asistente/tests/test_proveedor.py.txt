from django.test import SimpleTestCase, override_settings
from consulta.tests.utiles import asegurar_clinica_demo, asegurar_membresia, crear_clinica, crear_usuario_con_clinica

from asistente.proveedor import crear_modelo_chat


@override_settings(
    LLM_SERVER='openai',
    OPENAI_API_KEY='sk-test',
    OPENAI_MODEL='gpt-4.1-mini',
    OPENAI_REASONING_EFFORT='none',
)
class ProveedorOpenAiTests(SimpleTestCase):
    def test_cliente_openai_tiene_timeout_de_conexion(self):
        chat = crear_modelo_chat()
        cliente = chat.client._client
        self.assertIsNotNone(cliente.timeout.connect)
        self.assertLessEqual(cliente.timeout.connect, 15.0)
        self.assertGreaterEqual(cliente.timeout.read or 0, 60.0)
