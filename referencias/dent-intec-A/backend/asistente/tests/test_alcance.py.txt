from datetime import time, timedelta
from unittest.mock import patch

from django.contrib.auth import get_user_model
from django.core.management import call_command
from django.test import TestCase, override_settings
from django.utils import timezone
from consulta.tests.utiles import asegurar_clinica_demo, asegurar_membresia
from consulta.models import Cita, Paciente, Tratamiento

from asistente.alcance import (
    RESPUESTA_FUERA_DE_ALCANCE,
    clasificar_mensaje,
    respuesta_rapida,
)
from asistente.grafo import _mensajes_langchain
from asistente.models import Conversacion, Mensaje
from asistente.salida import validar_respuesta_asistente
from asistente.views import MensajesAsistenteView

User = get_user_model()


def _cita_manana_demo(clinica, *, inicio=time(9, 0)):
    """Una cita mañana en clínica demo (Juan) para plantillas de ruta rápida."""
    manana = timezone.localdate() + timedelta(days=1)
    Cita.objects.filter(clinica=clinica, fecha=manana).update(
        estado_asistencia=Cita.ASISTENCIA_CANCELADA
    )
    juan = Paciente.objects.get(clinica=clinica, nombre='Juan')
    cita = Cita.objects.create(
        clinica=clinica,
        paciente=juan,
        fecha=manana,
        inicio=inicio,
        bloques=1,
    )
    Tratamiento.objects.create(
        clinica=clinica,
        cita=cita,
        nombre='Limpieza dental',
        estado='programado',
        monto=500,
    )
    return cita


class AlcanceClasificacionTests(TestCase):
    def setUp(self):
        self.clinica = asegurar_clinica_demo()
        demo = User.objects.create_superuser('demo', 'demo@localhost', 'demo')
        asegurar_membresia(demo, clinica=self.clinica)
        call_command('cargar_demo')
        self.staff = User.objects.get(username='demo')
        self.conversacion = Conversacion.objects.create(
            usuario=self.staff,
            clinica=self.clinica,
            titulo='Prueba',
        )

    def test_rechaza_programacion(self):
        self.assertEqual(
            clasificar_mensaje('Dame un ejemplo de ciclo for en Python', self.conversacion),
            'rechazo',
        )

    def test_pasa_agenda(self):
        self.assertEqual(
            clasificar_mensaje('quien viene manana', self.conversacion),
            'agenda',
        )


class AlcanceVistaTests(TestCase):
    def setUp(self):
        self.clinica = asegurar_clinica_demo()
        demo = User.objects.create_superuser('demo', 'demo@localhost', 'demo')
        asegurar_membresia(demo, clinica=self.clinica)
        call_command('cargar_demo')
        self.staff = User.objects.get(username='demo')

    @patch('asistente.views.redactar_respuesta')
    def test_python_no_llama_llm(self, redactar_mock):
        from rest_framework.test import APIRequestFactory, force_authenticate

        factory = APIRequestFactory()
        request = factory.post(
            '/api/asistente/mensajes/',
            {'texto': 'Dame un ejemplo de ciclo for en Python'},
            format='json',
        )
        force_authenticate(request, user=self.staff)
        request.clinica = self.clinica
        response = MensajesAsistenteView.as_view()(request)
        self.assertEqual(response.status_code, 200)
        redactar_mock.assert_not_called()
        ultimo = response.data['mensajes'][-1]
        self.assertEqual(ultimo['texto'], RESPUESTA_FUERA_DE_ALCANCE)

    @override_settings(ASISTENTE_RUTA_RAPIDA=True)
    @patch('asistente.views.redactar_respuesta')
    def test_quien_viene_manana_ruta_rapida(self, redactar_mock):
        from rest_framework.test import APIRequestFactory, force_authenticate

        _cita_manana_demo(self.clinica)
        factory = APIRequestFactory()
        request = factory.post(
            '/api/asistente/mensajes/',
            {'texto': 'quien viene manana'},
            format='json',
        )
        force_authenticate(request, user=self.staff)
        request.clinica = self.clinica
        response = MensajesAsistenteView.as_view()(request)
        self.assertEqual(response.status_code, 200)
        redactar_mock.assert_not_called()
        texto = response.data['mensajes'][-1]['texto']
        self.assertIn('viene', texto.casefold())
        self.assertIn('Juan', texto)
        self.assertIn('9 AM', texto)


class HistorialFueraDeAlcanceTests(TestCase):
    def setUp(self):
        self.clinica = asegurar_clinica_demo()
        demo = User.objects.create_superuser('demo', 'demo@localhost', 'demo')
        asegurar_membresia(demo, clinica=self.clinica)
        self.staff = User.objects.get(username='demo')
        self.conversacion = Conversacion.objects.create(
            usuario=self.staff,
            clinica=self.clinica,
            titulo='Prueba',
        )

    def test_omite_par_rechazado_en_historial(self):
        Mensaje.objects.create(
            conversacion=self.conversacion,
            clinica=self.clinica,
            rol=Mensaje.Rol.USUARIO,
            texto='Dame codigo python',
        )
        Mensaje.objects.create(
            conversacion=self.conversacion,
            clinica=self.clinica,
            rol=Mensaje.Rol.ASISTENTE,
            texto=RESPUESTA_FUERA_DE_ALCANCE,
            fuera_de_alcance=True,
        )
        Mensaje.objects.create(
            conversacion=self.conversacion,
            clinica=self.clinica,
            rol=Mensaje.Rol.USUARIO,
            texto='quien viene hoy',
        )
        historial = _mensajes_langchain(self.conversacion)
        textos = [m.content for m in historial if hasattr(m, 'content')]
        self.assertNotIn('Dame codigo python', textos)
        self.assertIn('quien viene hoy', textos)


class SalidaValidacionTests(TestCase):
    def test_rechaza_bloque_codigo(self):
        texto = validar_respuesta_asistente('```python\nfor x in range(3):\n    print(x)\n```')
        self.assertEqual(texto, RESPUESTA_FUERA_DE_ALCANCE)

    def test_convierte_hora_24h(self):
        self.assertIn('9 AM', validar_respuesta_asistente('La cita es a las 09:00.'))


class RespuestaRapidaTests(TestCase):
    def setUp(self):
        self.clinica = asegurar_clinica_demo()
        demo = User.objects.create_superuser('demo', 'demo@localhost', 'demo')
        asegurar_membresia(demo, clinica=self.clinica)
        call_command('cargar_demo')

    def test_plantilla_manana(self):
        _cita_manana_demo(self.clinica)
        texto = respuesta_rapida('quien viene manana', self.clinica)
        self.assertIsNotNone(texto)
        self.assertIn('viene', texto.casefold())
        self.assertIn('Juan', texto)
        self.assertIn('9 AM', texto)
