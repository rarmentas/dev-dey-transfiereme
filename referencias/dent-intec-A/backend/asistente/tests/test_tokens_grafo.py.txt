from unittest.mock import MagicMock

from django.test import SimpleTestCase
from langchain_core.messages import AIMessage, HumanMessage

from asistente.contexto import establecer_conversacion, restablecer_conversacion
from asistente.grafo import _invocar_con_herramientas


class TokensGrafoLogTests(SimpleTestCase):
    def _historial_minimo(self):
        return [HumanMessage(content='hola')]

    def test_log_con_usage_metadata(self):
        chat = MagicMock()
        salida = AIMessage(content='respuesta')
        salida.usage_metadata = {
            'input_tokens': 10,
            'output_tokens': 5,
            'total_tokens': 15,
        }
        chat.invoke.return_value = salida
        with self.assertLogs('asistente.grafo', level='INFO') as capturado:
            texto = _invocar_con_herramientas(self._historial_minimo(), chat=chat)
        self.assertEqual(texto, 'respuesta')
        self.assertEqual(len(capturado.records), 1)
        mensaje = capturado.output[0]
        self.assertIn('tokens_llm', mensaje)
        self.assertIn('conversacion=None', mensaje)
        self.assertIn('ronda=1', mensaje)
        self.assertIn('input_tokens=10', mensaje)
        self.assertIn('output_tokens=5', mensaje)
        self.assertIn('total_tokens=15', mensaje)

    def test_log_sin_usage_metadata(self):
        chat = MagicMock()
        chat.invoke.return_value = AIMessage(content='ok')
        with self.assertLogs('asistente.grafo', level='INFO') as capturado:
            _invocar_con_herramientas(self._historial_minimo(), chat=chat)
        mensaje = capturado.output[0]
        self.assertIn('input_tokens=None', mensaje)
        self.assertIn('output_tokens=None', mensaje)
        self.assertIn('total_tokens=None', mensaje)

    def test_log_incluye_id_conversacion(self):
        conversacion = MagicMock()
        conversacion.pk = 42
        token = establecer_conversacion(conversacion)
        try:
            chat = MagicMock()
            chat.invoke.return_value = AIMessage(content='listo')
            with self.assertLogs('asistente.grafo', level='INFO') as capturado:
                _invocar_con_herramientas(self._historial_minimo(), chat=chat)
            self.assertIn('conversacion=42', capturado.output[0])
        finally:
            restablecer_conversacion(token)
